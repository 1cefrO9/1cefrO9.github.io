 6// 构建KMP算法的next数组
 7vector<int> buildNext(const string& pattern) {
 8    int n = pattern.size();
 9    vector<int> next(n, 0);    // next数组初始化为0
10    int j = 0;                 // 前缀指针
11
12    for (int i = 1; i < n; i++) { // 后缀指针从1开始
13        while (j > 0 && pattern[i] != pattern[j]) {
14            j = next[j - 1];      // 回溯到前一个匹配位置
15        }
16        if (pattern[i] == pattern[j]) {
17            j++;                  // 匹配成功，前后缀指针同时后移
18        }
19        next[i] = j;              // 记录当前位置的最长公共前后缀长度
20    }
21    return next;
22}
23
24// KMP主匹配函数
25vector<int> kmpSearch(const string& text, const string& pattern) {
26    vector<int> matches;          // 存储所有匹配的起始位置
27    if (pattern.empty()) return matches;
28
29    vector<int> next = buildNext(pattern);
30    int j = 0;                    // 模式串指针
31    int m = text.size(), n = pattern.size();
32
33    for (int i = 0; i < m; i++) { // 遍历文本串
34        while (j > 0 && text[i] != pattern[j]) {
35            j = next[j - 1];      // 利用next数组跳过不必要的比较
36        }
37        if (text[i] == pattern[j]) {
38            j++;
39        }
40        if (j == n) {             // 完全匹配成功
41            matches.push_back(i - n + 1);
42            j = next[j - 1];       // 继续寻找下一个可能的匹配
43        }
44    }
45    return matches;
46}
